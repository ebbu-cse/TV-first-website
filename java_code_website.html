import java.util.Random;

// Class for even numbers
class EvenNum implements Runnable {
    int a;

    public EvenNum(int a) {
        this.a = a;
    }

    public void run() {
        System.out.println(
            "Thread " + a + " is EVEN and square is: " + (a * a)
        );
    }
}

// Class for odd numbers
class OddNum implements Runnable {
    int a;

    public OddNum(int a) {
        this.a = a;
    }

    public void run() {
        System.out.println(
            "Thread " + a + " is ODD and cube is: " + (a * a * a)
        );
    }
}

// Class to generate random numbers
class RandomNumGenerator extends Thread {
    public void run() {
        Random rand = new Random();

        try {
            for (int i = 0; i < 10; i++) {
                int n = rand.nextInt(20);

                System.out.println("Generated Number is: " + n);

                if (n % 2 == 0) {
                    Thread t1 = new Thread(new EvenNum(n));
                    t1.start();
                    t1.join();
                } else {
                    Thread t2 = new Thread(new OddNum(n));
                    t2.start();
                    t2.join();
                }

                Thread.sleep(1000);
                System.out.println("--------------------");
            }
        } catch (InterruptedException ex) {
            System.out.println(ex.getMessage());
            Thread.currentThread().interrupt();
        }
    }
}

// Driver class
public class MultiThreadRandOddEven {
    public static void main(String[] args) {
        RandomNumGenerator randNum = new RandomNumGenerator();
        randNum.start();
    }
}
